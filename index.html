Petunjuk Penggunaan & Kode Backend (Code.gs)

Langkah Mudah Pemasangan di Google Apps Script:
  1. Buka Spreadsheet Anda: Klik Tautan Spreadsheet
  2. Klik menu Extensions (Ekstensi) > Apps Script.
  3. Hapus kode bawaan di `Code.gs`, lalu salin seluruh kode di bawah ini.
  4. Klik tombol Deploy (Terapkan) > New Deployment (Penerapan Baru).
  5. Pilih jenis Web App, atur Execute as: Me dan Who has access: Anyone.
  6. Salin URL Web App yang dihasilkan, lalu masukkan ke kolom konfigurasi di Portal Admin web ini.
Kode Skrip Backend (Code.gs): 
// KODE BACKEND GOOGLE APPS SCRIPT (Code.gs) - PT. JALUR DATA INDONESIA
const SPREADSHEET_ID = "1bR4QVxYM1sTtdaNgCS3uNA9vQHuw7iCL2cXu_DDN3TM";
const SHEET_NAME = "PelangganBaru";

function doGet(e) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = getOrCreateSheet(ss, SHEET_NAME);
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const rows = [];
    
    for (let i = 1; i < data.length; i++) {
      let rowObj = {};
      for (let j = 0; j < headers.length; j++) {
        rowObj[headers[j]] = data[i][j];
      }
      rows.push(rowObj);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", data: rows }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = getOrCreateSheet(ss, SHEET_NAME);
    const body = JSON.parse(e.postData.contents);
    
    const timestamp = new Date();
    sheet.appendRow([
      timestamp,
      body.reservationNumber || "",
      body.name || "",
      body.phone || "",
      body.desa || "",
      body.package || "",
      body.address || "",
      body.status || "Menunggu Verifikasi",
      body.photoData ? "Terlampir" : "Tidak ada foto"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ 
      status: "success", 
      message: "Data pelanggan berhasil disimpan di Spreadsheet PT JDI!",
      reservationNumber: body.reservationNumber 
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet(ss, name) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow([
      "Waktu Daftar", "No. Reservasi", "Nama Lengkap", "No. WhatsApp", 
      "Wilayah / Desa", "Paket Internet", "Alamat & Patokan", "Status", "Keterangan Foto"
    ]);
    sheet.getRange(1, 1, 1, 9).setFontWeight("bold").setBackground("#0891b2").setFontColor("#ffffff");
  }
  return sheet;
}
